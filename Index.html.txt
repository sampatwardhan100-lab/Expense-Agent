    $("authToggleBtn").textContent = isSignUpMode ? "Sign In" : "Create an account";
    $("authError").textContent = "";
  };

  // Auth Submission
  $("authForm").onsubmit = async e => {
    e.preventDefault();
    const email = $("authEmail").value.trim();
    const password = $("authPassword").value;
    $("authSubmitBtn").disabled = true;
    $("authError").textContent = "";

    try {
      if (isSignUpMode) {
        await createUserWithEmailAndPassword(auth, email, password);
      } else {
        await signInWithEmailAndPassword(auth, email, password);
      }
    } catch (err) {
      $("authError").textContent = err.message.replace("Firebase: ", "");
    } finally {
      $("authSubmitBtn").disabled = false;
    }
  };

  $("signOutBtn").onclick = () => signOut(auth);

  // Monitor Authentication State
  onAuthStateChanged(auth, user => {
    currentUser = user;
    if (user) {
      $("authSection").style.display = "none";
      $("appSection").style.display = "block";
      $("userEmailDisplay").textContent = user.email;
      bindFirestore(user.uid);
    } else {
      if (unsubscribeSnapshot) {
        unsubscribeSnapshot();
        unsubscribeSnapshot = null;
      }
      S.items = [];
      $("authSection").style.display = "block";
      $("appSection").style.display = "none";
    }
  });

  // Bind Firestore to User Subcollection
  function bindFirestore(uid) {
    const userExpensesRef = collection(db, "users", uid, "expenses");
    $("statusDot").className = "status-dot";
    $("statusTxt").textContent = "Syncing…";

    unsubscribeSnapshot = onSnapshot(userExpensesRef, snapshot => {
      const cloudItems = [];
      snapshot.forEach(docSnap => cloudItems.push({ id: docSnap.id, ...docSnap.data() }));
      cloudItems.sort((a, b) => (b.date || "").localeCompare(a.date || "") || (b.createdAt || 0) - (a.createdAt || 0));
      S.items = cloudItems;
      $("statusDot").className = "status-dot ok";
      $("statusTxt").textContent = "Cloud Synced";
      render();
    }, err => {
      console.error("Firestore listener error:", err);
      $("statusDot").className = "status-dot err";
      $("statusTxt").textContent = err.code === "permission-denied" ? "Rules Blocked" : "Sync Error";
    });
  }

  // Parse Natural Language
  function parseExpenseText(text) {
    let baseDate = iso(new Date());
    if (/day before yesterday/i.test(text)) {
      const d = new Date(); d.setDate(d.getDate() - 2); baseDate = iso(d);
    } else if (/yesterday/i.test(text)) {
      const d = new Date(); d.setDate(d.getDate() - 1); baseDate = iso(d);
    }

    let defaultPay = "";
    if (/upi|gpay|phonepe|paytm/i.test(text)) defaultPay = "UPI";
    else if (/card|credit|debit/i.test(text)) defaultPay = "Card";
    else if (/cash/i.test(text)) defaultPay = "Cash";

    const parts = text.split(/\s*(?:,|\band\b|\n|;)\s*/i).filter(Boolean);
    const results = [];

    for (const part of parts) {
      const m = part.match(/(\d[\d,]*(?:\.\d+)?)\s*(k)?/i);
      if (!m) continue;
      let amt = parseFloat(m[1].replace(/,/g, ""));
      if (m[2]) amt *= 1000;
      if (!amt || isNaN(amt)) continue;

      let itemPay = defaultPay;
      if (/upi|gpay|phonepe|paytm/i.test(part)) itemPay = "UPI";
      else if (/card|credit|debit/i.test(part)) itemPay = "Card";
      else if (/cash/i.test(part)) itemPay = "Cash";

      let itemDate = baseDate;
      if (/day before yesterday/i.test(part)) {
        const d = new Date(); d.setDate(d.getDate() - 2); itemDate = iso(d);
      } else if (/yesterday/i.test(part)) {
        const d = new Date(); d.setDate(d.getDate() - 1); itemDate = iso(d);
      } else if (/today/i.test(part)) {
        itemDate = iso(new Date());
      }

      let desc = part
        .replace(m[0], "")
        .replace(/\b(today|yesterday|day before yesterday|spent|on|for|paid|via|by|rs|inr|upi|gpay|cash|card)\b|[₹$€£]/gi, "")
        .trim() || "Expense";

      desc = desc.charAt(0).toUpperCase() + desc.slice(1);

      let cat = "Other";
      for (const c in KW) {
        if (KW[c].test(part)) { cat = c; break; }
      }

      results.push({
        amount: Math.round(amt * 100) / 100,
        category: cat,
        description: desc.slice(0, 50),
        payment: itemPay,
        date: itemDate,
        createdAt: Date.now()
      });
    }
    return results;
  }

  // Add Expenses
  async function handleAdd() {
    if (!currentUser) return;
    const val = $("expInput").value.trim();
    if (!val) return;

    $("addBtn").disabled = true;
    $("helpMsg").textContent = "Processing…";

    const parsed = parseExpenseText(val);
    if (!parsed.length) {
      $("helpMsg").textContent = "Couldn’t detect an amount. Example: 'Lunch 350, metro 40'";
      $("addBtn").disabled = false;
      return;
    }

    const userExpensesRef = collection(db, "users", currentUser.uid, "expenses");
    try {
      for (const item of parsed) {
        await addDoc(userExpensesRef, item);
      }
      $("expInput").value = "";
      $("helpMsg").textContent = `Saved ${parsed.length} expense${parsed.length > 1 ? "s" : ""}!`;
      setTimeout(() => { $("helpMsg").textContent = ""; }, 3000);
    } catch (err) {
      $("helpMsg").textContent = "Failed to save: " + err.message;
    } finally {
      $("addBtn").disabled = false;
    }
  }

  async function handleDelete(id) {
    if (!currentUser) return;
    try {
      await deleteDoc(doc(db, "users", currentUser.uid, "expenses", id));
    } catch (err) {
      console.error("Delete failed:", err);
    }
  }

  function render() {
    $("curSelector").value = S.cur;
    $("monthLabel").textContent = parseMonth(viewMonth).toLocaleString(undefined, { month: "long", year: "numeric" });

    const monthItems = S.items.filter(i => (i.date || "").startsWith(viewMonth));
    const total = monthItems.reduce((acc, i) => acc + (Number(i.amount) || 0), 0);
    $("totalDisplay").textContent = fmt(total);

    const byCat = {};
    monthItems.forEach(i => { byCat[i.category] = (byCat[i.category] || 0) + Number(i.amount); });
    const sortedCats = Object.entries(byCat).sort((a, b) => b[1] - a[1]);
    const maxCatVal = sortedCats.length ? sortedCats[0][1] : 1;

    $("catBreakdown").innerHTML = sortedCats.length
      ? sortedCats.map(([c, v]) => `
          <div class="bar">
            <span class="cat">${c}</span>
            <span class="track"><i style="width: ${(v / maxCatVal) * 100}%"></i></span>
            <span class="val">${fmt(v)}</span>
          </div>`).join("")
      : '<div style="color:var(--mu);font-size:13px;padding:6px 0;">No expenses logged for this month yet.</div>';

    const months = [];
    for (let k = 5; k >= 0; k--) {
      const d = parseMonth(viewMonth);
      d.setMonth(d.getMonth() - k);
      months.push(iso(d).slice(0, 7));
    }
    const trendTotals = months.map(m => S.items.filter(i => (i.date || "").startsWith(m)).reduce((a, b) => a + (Number(b.amount) || 0), 0));
    const maxTrend = Math.max(...trendTotals, 1);

    $("trendContainer").innerHTML = months.map((m, idx) => `
      <div class="trend-col">
        <div class="trend-bar-box">
          <i class="${m === viewMonth ? 'active' : ''}" style="height: ${Math.max((trendTotals[idx] / maxTrend) * 100, 4)}%" data-m="${m}"></i>
        </div>
        ${parseMonth(m).toLocaleString(undefined, { month: "short" })}
      </div>
    `).join("");

    document.querySelectorAll(".trend-bar-box i").forEach(el => {
      el.onclick = () => { viewMonth = el.dataset.m; render(); };
    });

    const listEl = $("expenseList");
    if (!monthItems.length) {
      listEl.innerHTML = '<div class="empty-state">No entries for this month. Type an expense above!</div>';
      return;
    }

    listEl.innerHTML = "";
    monthItems.forEach(item => {
      const row = document.createElement("div");
      row.className = "item";
      const tags = [item.category, item.payment, item.date].filter(Boolean).join(" · ");
      row.innerHTML = `
        <div class="item-info">
          <b>${item.description}</b>
          <span>${tags}</span>
        </div>
        <div class="item-amount">${fmt(item.amount)}</div>
        <button class="btn-del" title="Delete">×</button>
      `;
      row.querySelector(".btn-del").onclick = () => handleDelete(item.id);
      listEl.appendChild(row);
    });
  }

  $("addBtn").onclick = handleAdd;
  $("expInput").addEventListener("keydown", e => {
    if (e.key === "Enter" && !e.shiftKey) {
      e.preventDefault();
      handleAdd();
    }
  });

  $("curSelector").onchange = e => { S.cur = e.target.value; render(); };
  $("prevMonth").onclick = () => {
    const d = parseMonth(viewMonth);
    d.setMonth(d.getMonth() - 1);
    viewMonth = iso(d).slice(0, 7);
    render();
  };
  $("nextMonth").onclick = () => {
    const d = parseMonth(viewMonth);
    d.setMonth(d.getMonth() + 1);
    viewMonth = iso(d).slice(0, 7);
    render();
  };

  const downloadBlob = (filename, content, type) => {
    const blob = new Blob([content], { type });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = filename;
    a.click();
    URL.revokeObjectURL(url);
  };

  $("btnCsv").onclick = () => {
    if (!S.items.length) return;
    const q = s => `"${String(s || '').replace(/"/g, '""')}"`;
    const rows = [
      ["Date", "Category", "Description", "Payment", "Amount"],
      ...S.items.map(i => [i.date, i.category, q(i.description), i.payment || "", i.amount])
    ];
    downloadBlob(`expenses_${viewMonth}.csv`, rows.map(r => r.join(",")).join("\n"), "text/csv");
  };

  $("btnBackup").onclick = () => {
    downloadBlob("expenses_backup.json", JSON.stringify(S.items, null, 2), "application/json");
  };

  $("btnRestore").onclick = () => $("fileInput").click();
  $("fileInput").onchange = async e => {
    const file = e.target.files[0];
    if (!file || !currentUser) return;
    try {
      const items = JSON.parse(await file.text());
      const userExpensesRef = collection(db, "users", currentUser.uid, "expenses");
      if (Array.isArray(items)) {
        for (const item of items) {
          const { id, ...data } = item;
          await addDoc(userExpensesRef, data);
        }
      }
    } catch (err) {
      console.error("Restore failed:", err);
    }
    e.target.value = "";
  };
